cmake_minimum_required(VERSION 3.16)
project(uqm_conversation_design LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(nlohmann_json 3.2 REQUIRED)
find_package(Boost REQUIRED COMPONENTS unit_test_framework)
set(DIALOGUE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../sc2/src/uqm/dialogue")
add_executable(conversation-design-tests ${DIALOGUE_SOURCE}/engine.cpp ${DIALOGUE_SOURCE}/graph.cpp ${DIALOGUE_SOURCE}/verb_factory.cpp tests.cpp)
target_link_libraries(conversation-design-tests PRIVATE nlohmann_json::nlohmann_json Boost::unit_test_framework)
target_compile_definitions(conversation-design-tests PRIVATE BOOST_TEST_DYN_LINK DESIGN_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(conversation-design-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME conversation-design COMMAND conversation-design-tests)
